import Foundation

func movie(_ id: String, _ minutes: Double?) -> BaseItem {
    var item = BaseItem(id: id, name: "Movie \(id)", type: "Movie")
    item.runTimeTicks = minutes.map { Int64($0 * 60 * 10_000_000) }
    let n = Int(id.dropFirst())!; item.premiereDate = "\(1980 + n)-0\(1 + n % 9)-15T00:00:00.0000000Z"
    return item
}
func episode(_ series: String, _ season: Int, _ number: Int, _ minutes: Double = 22) -> BaseItem {
    var item = BaseItem(id: "\(series)-s\(season)e\(number)", name: "\(series) \(season)x\(number)", type: "Episode")
    item.seriesId = series
    item.seriesName = series.capitalized
    item.parentIndexNumber = season
    item.indexNumber = number
    item.runTimeTicks = Int64(minutes * 60 * 10_000_000)
    item.premiereDate = "\(2000 + number + season)-0\(1 + (number + season) % 9)-10T00:00:00.0000000Z"
    return item
}
func clip(_ id: String, _ seconds: Double, tags: [String]? = nil) -> BaseItem {
    var item = BaseItem(id: id, name: id, type: "Movie")
    item.runTimeTicks = Int64(seconds * 10_000_000)
    item.tags = tags
    return item
}

func zone(_ id: String) -> Calendar {
    var c = Calendar(identifier: .gregorian); c.timeZone = TimeZone(identifier: id)!; return c
}
func date(_ cal: Calendar, _ y: Int, _ m: Int, _ d: Int, _ h: Int = 0, _ mi: Int = 0) -> Date {
    cal.date(from: DateComponents(year: y, month: m, day: d, hour: h, minute: mi))!
}

struct Scenario {
    var name: String
    var channel = "channel-a"
    var tz = "America/Chicago"
    var items: [BaseItem] = movies
    var lineup = ChannelLineup()
    var clips: ChannelBreakClips = .none
    var window: (Int, Int, Int, Int, Int, Int, Int, Int, Int, Int) // y m d h mi  y m d h mi
}
func w(_ a: (Int, Int, Int, Int), _ b: (Int, Int, Int, Int)) -> (Int, Int, Int, Int, Int, Int, Int, Int, Int, Int) { (a.0, a.1, a.2, a.3, 0, b.0, b.1, b.2, b.3, 0) }

nonisolated(unsafe) var scenarios: [Scenario] = []
let day = w((2026, 10, 15, 18), (2026, 10, 17, 6))
func add(_ name: String, _ items: [BaseItem], _ configure: (inout ChannelLineup) -> Void = { _ in }, clips: ChannelBreakClips = .none, channel: String = "channel-a", tz: String = "America/Chicago", window: (Int, Int, Int, Int, Int, Int, Int, Int, Int, Int)? = nil) {
    var lineup = ChannelLineup(); configure(&lineup)
    scenarios.append(Scenario(name: name, channel: channel, tz: tz, items: items, lineup: lineup, clips: clips, window: window ?? day))
}

func f(_ d: Date) -> String { String(format: "%.3f", d.timeIntervalSince1970) }
func roleName(_ r: ProgramRole) -> String { "\(r)" }

var out = "["
for (index, s) in scenarios.enumerated() {
    let cal = zone(s.tz)
    let from = date(cal, s.window.0, s.window.1, s.window.2, s.window.3, s.window.4)
    let to = date(cal, s.window.5, s.window.6, s.window.7, s.window.8, s.window.9)
    let segs = ChannelSchedule.segments(channelId: s.channel, items: s.items, lineup: s.lineup, breaks: s.clips, from: from, to: to, calendar: cal)
    let progs = ChannelSchedule.programs(channelId: s.channel, items: s.items, lineup: s.lineup, breaks: s.clips, from: from, to: to, calendar: cal)
    func list(_ ps: [ScheduledProgram]) -> String {
        "[" + ps.map { "[\(f($0.start)),\(f($0.end)),\"\($0.item.id)\",\"\(roleName($0.role))\"]" }.joined(separator: ",") + "]"
    }
    out += (index > 0 ? "," : "") + "\n{\"name\":\"\(s.name)\",\"tz\":\"\(s.tz)\",\"from\":\(f(from)),\"to\":\(f(to)),\"segments\":\(list(segs)),\"programs\":\(list(progs))}"
}
out += "\n]"
print(out)
